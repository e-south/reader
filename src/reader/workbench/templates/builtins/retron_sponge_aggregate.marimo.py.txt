    retron_aggregate_bundle_error = None
    try:
        retron_aggregate_bundle = load_retron_review_bundle(
            retron_aggregate_manifest_path,
            source_root=retron_aggregate_manifest_path.parent.parent,
        )
    except Exception as retron_aggregate_exc:
        retron_aggregate_bundle_error = str(retron_aggregate_exc)

        mo.stop(

                f"`{retron_aggregate_manifest_path}` could not be resolved.\n\n"
                f"{retron_aggregate_bundle_error}"
            ),
        )
    return (retron_aggregate_bundle,)

@app.cell(hide_code=True)
def _(
    mo,
    retron_aggregate_architecture_selector,
    retron_aggregate_control_panel,
    retron_aggregate_contextual_copy,
    retron_aggregate_expected_selector,
    retron_aggregate_result,
    retron_aggregate_result_error,
    retron_aggregate_score_selector,
    retron_aggregate_selected_plot_id,
    retron_aggregate_export_panel,
    retron_aggregate_supporting_table_preview,
    retron_table_kwargs,
):
    retron_aggregate_review_blocks = [
        retron_aggregate_control_panel,
    ]
    if retron_aggregate_result_error:
        retron_aggregate_review_blocks.append(
            mo.callout(
                mo.md(retron_aggregate_result_error),
                kind="danger",
            )
        )
    elif retron_aggregate_result is None:
        retron_aggregate_review_blocks.append(
            mo.callout(
                mo.md("The selected aggregate plot did not resolve a render result."),
                kind="warn",
            )
        )
    elif retron_aggregate_result.figure is None:
        retron_aggregate_review_blocks.append(
            mo.callout(
                mo.md("No figure is available for the current selection."),
                kind="warn",
            )
        )
    else:
        retron_aggregate_review_blocks.append(retron_aggregate_result.figure)
    if retron_aggregate_result is not None and retron_aggregate_contextual_copy is not None:
        retron_aggregate_review_blocks.append(
            mo.callout(
                mo.vstack(
                    [
                        mo.md(f"### {retron_aggregate_result.title}"),
                        mo.md(f"**Question**\n{retron_aggregate_contextual_copy['question']}"),
                        mo.md(f"**Formula**\n```text\n{retron_aggregate_contextual_copy['math']}\n```"),
                        mo.md(f"**Interpretation**\n{retron_aggregate_contextual_copy['meaning']}"),
                    ],
                    gap=0.08,
                ),
                kind="info",
            )
        )
    if retron_aggregate_export_panel is not None:
        retron_aggregate_review_blocks.append(retron_aggregate_export_panel)

    if retron_aggregate_result is not None:
        retron_aggregate_review_blocks.append(
            mo.ui.table(
                retron_aggregate_supporting_table_preview,
                **retron_table_kwargs(
                    page_size=15,
                    max_height=420,
                ),
            )
        )
    retron_aggregate_review_panel = mo.vstack(retron_aggregate_review_blocks, gap=0.45)
    return (retron_aggregate_review_panel,)

        if retron_source_plot_selector_options
        else None

@app.cell(hide_code=True)
def _(
    dataframe_to_csv_bytes,
    download_safe_stem,
    figure_to_download_bytes,
    mo,
    retron_source_plot_result,
    retron_source_plot_selector,
    retron_source_selected_figure,
    retron_source_selected_source,
    retron_source_supporting_table,
):
    retron_source_export_panel = None
    if (
        retron_source_plot_result is not None
        and retron_source_plot_selector is not None
        and retron_source_selected_source is not None
        and retron_source_supporting_table is not None
    ):
        retron_source_export_stem = download_safe_stem(
            f"{retron_source_selected_source.label}__{retron_source_plot_selector.value}"
        )
        retron_source_export_items = [
            mo.download(
                data=lambda df=retron_source_supporting_table: dataframe_to_csv_bytes(df),
                filename=f"{retron_source_export_stem}__supporting_table.csv",
                mimetype="text/csv",
                label="Table CSV",
            )
        ]
        if retron_source_selected_figure is not None:
            retron_source_export_items = [
                mo.download(
                    data=lambda fig=retron_source_selected_figure: figure_to_download_bytes(
                        fig,
                        fmt="png",
                    ),
                    filename=f"{retron_source_export_stem}.png",
                    mimetype="image/png",
                    label="Figure PNG",
                ),
                mo.download(
                    data=lambda fig=retron_source_selected_figure: figure_to_download_bytes(
                        fig,
                        fmt="svg",
                    ),
                    filename=f"{retron_source_export_stem}.svg",
                    mimetype="image/svg+xml",
                    label="Figure SVG",
                ),
                mo.download(
                    data=lambda fig=retron_source_selected_figure: figure_to_download_bytes(
                        fig,
                        fmt="pdf",
                    ),
                    filename=f"{retron_source_export_stem}.pdf",
                    mimetype="application/pdf",
                    label="Figure PDF",
                ),
                *retron_source_export_items,
            ]
        retron_source_export_panel = mo.hstack(
            retron_source_export_items,
            justify="start",
            wrap=True,
            gap=0.5,
        )
    return (retron_source_export_panel,)


@app.cell(hide_code=True)
def _(
    mo,
    retron_source_contextual_copy,
    retron_source_context_tabs,
    retron_source_control_panel,
    retron_source_export_panel,
    retron_source_plot_error,
    retron_source_plot_result,
    retron_source_selected_figure,
    retron_source_selected_source,
    retron_source_supporting_table_preview,
    retron_source_surface_error,
    retron_source_supporting_table,
    retron_table_kwargs,
):
    retron_source_review_blocks = []
    if retron_source_context_tabs is not None:
        retron_source_review_blocks.append(retron_source_context_tabs)
    retron_source_review_blocks.append(retron_source_control_panel)
    if retron_source_surface_error:
        retron_source_review_blocks.append(
            mo.callout(
                mo.md(
                    "This source experiment cannot be opened as a scoped review surface from the "
                    "master dashboard.\n\n"
                    f"{retron_source_surface_error}"
                ),
                kind="warn",
            )
        )
    elif retron_source_plot_error:
        retron_source_review_blocks.append(
            mo.callout(
                mo.md(retron_source_plot_error),
                kind="warn",
            )
        )
    elif retron_source_plot_result is None:
        retron_source_review_blocks.append(
            mo.callout(
                mo.md("No source-scoped plot is available for the current selection."),
                kind="warn",
            )
        )
    else:
        if retron_source_selected_figure is None:
            retron_source_review_blocks.append(
                mo.callout(
                    mo.md("The selected source plot did not resolve a figure."),
                    kind="warn",
                )
            )
        else:
            retron_source_review_blocks.append(retron_source_selected_figure)
        retron_source_review_blocks.append(
            mo.callout(
                mo.vstack(
                    [
                        mo.md(f"### {retron_source_plot_result.title}"),
                        mo.md(f"**Question**\n{retron_source_contextual_copy['question']}"),
                        mo.md(f"**Formula**\n```text\n{retron_source_contextual_copy['math']}\n```"),
                        mo.md(f"**Interpretation**\n{retron_source_contextual_copy['meaning']}"),
                    ],
                    gap=0.08,
                ),
                kind="info",
            )
        )
        if retron_source_export_panel is not None:
            retron_source_review_blocks.append(retron_source_export_panel)
        if retron_source_supporting_table is not None:
            retron_source_review_blocks.append(
                mo.ui.table(
                    retron_source_supporting_table_preview,
                    **retron_table_kwargs(
                        page_size=15,
                        max_height=420,
                    ),
                )
            )
    retron_source_review_panel = mo.vstack(retron_source_review_blocks, gap=0.45)
    return (retron_source_review_panel,)
