@app.cell(hide_code=True)
def _(mo, plt, retron_review_architecture_df, retron_review_architecture_x, retron_review_score_metric):
    retron_review_architecture_output = None
    if retron_review_architecture_df.empty:
        retron_review_architecture_output = mo.md("No architecture-plot data are available.")
    else:
        _retron_review_architecture_sensors = sorted(
            retron_review_architecture_df["sensor"].dropna().astype(str).unique()
        )
        _retron_review_fig, _retron_review_axes = plt.subplots(
            1,
            len(_retron_review_architecture_sensors),
            figsize=(5.0 * len(_retron_review_architecture_sensors), 4.5),
            constrained_layout=True,
            squeeze=False,
        )
        for _retron_review_axis, _retron_review_sensor in zip(
            _retron_review_axes[0], _retron_review_architecture_sensors, strict=True
        ):
            _retron_review_sensor_df = retron_review_architecture_df[
                retron_review_architecture_df["sensor"].astype(str) == _retron_review_sensor
            ].copy()
            _retron_review_axis.scatter(
                _retron_review_sensor_df[retron_review_architecture_x.value],
                _retron_review_sensor_df["value"],
                s=70,
                alpha=0.85,
                edgecolors="black",
                linewidths=0.5,
            )
            for _, _retron_review_row in _retron_review_sensor_df.iterrows():
                _retron_review_axis.annotate(
                    str(_retron_review_row["sponge"]),
                    (
                        float(_retron_review_row[retron_review_architecture_x.value]),
                        float(_retron_review_row["value"]),
                    ),
                    xytext=(4, 4),
                    textcoords="offset points",
                )
            _retron_review_axis.axhline(0.0, color="#777777", linestyle=":", linewidth=1.0)
            _retron_review_axis.set_title(str(_retron_review_sensor))
            _retron_review_axis.set_xlabel(retron_review_architecture_x.value)
            _retron_review_axis.set_ylabel(retron_review_score_metric.value)
        retron_review_architecture_output = _retron_review_fig

                edgecolors="black",

            _retron_review_min = min(float(_retron_review_x.min()), float(_retron_review_y.min()))
            _retron_review_max = max(float(_retron_review_x.max()), float(_retron_review_y.max()))
            _retron_review_axis.plot(
                [_retron_review_min, _retron_review_max],
                [_retron_review_min, _retron_review_max],
                linestyle=":",
                color="#777777",
                linewidth=1.0,
            )
            for _, _retron_review_row in _retron_review_sensor_df.iterrows():

                    str(_retron_review_row["sponge"]),

                        float(_retron_review_row[retron_review_expected_mode.value]),
