            retron_source_plot_memory[retron_source_plot_key] = retron_source_plot_selector.value
            retron_set_source_plot_memory(retron_source_plot_memory)
        if retron_get_source_last_plot() != retron_source_plot_selector.value:
            retron_set_source_last_plot(retron_source_plot_selector.value)

    if retron_source_selected_source is not None and retron_source_plot_selector is not None:

@app.cell(hide_code=True)
def _(
    build_label_value_options,
    mo,
    retron_figure_option_rows,
    retron_get_source_figure_memory,
    retron_source_plot_result,
    retron_source_plot_selector,
    retron_source_selected_source,
):
    retron_source_figure_option_rows = (
        retron_figure_option_rows(retron_source_plot_result.figures) if retron_source_plot_result is not None else []
    )
    retron_source_figure_options = build_label_value_options(
        retron_source_figure_option_rows,
        label_key="Label",
        value_key="Filename",
        disambiguator_key="Filename",
    )
    retron_source_figure_selected_key = None
    if (
        retron_source_selected_source is not None
        and retron_source_plot_selector is not None
        and retron_source_figure_options
    ):
        retron_source_figure_memory_key = (
            f"{retron_source_selected_source.experiment_id}::{retron_source_plot_selector.value}"
        )
        retron_source_remembered_filename = retron_get_source_figure_memory().get(retron_source_figure_memory_key)
        retron_source_figure_selected_key = next(
            (key for key, value in retron_source_figure_options.items() if value == retron_source_remembered_filename),
            next(iter(retron_source_figure_options), None),
        )
    retron_source_figure_selector = (
        mo.ui.dropdown(
            options=retron_source_figure_options,
            value=retron_source_figure_selected_key,
            label="Figure scope",
            full_width=False,
        )
        if len(retron_source_figure_options) > 1
        else None
    )
    return retron_source_figure_option_rows, retron_source_figure_options, retron_source_figure_selector


@app.cell(hide_code=True)
def _(retron_source_figure_selector, retron_source_plot_result):
    retron_source_selected_figure = None
    retron_source_selected_filename = None
    if retron_source_plot_result is not None and retron_source_plot_result.figures:
        retron_source_selected_filename = (
            retron_source_figure_selector.value
            if retron_source_figure_selector is not None
            else retron_source_plot_result.figures[0].filename
        )
        for retron_source_figure in retron_source_plot_result.figures:
            if retron_source_figure.filename == retron_source_selected_filename:
                retron_source_selected_figure = retron_source_figure.fig
                break
    return retron_source_selected_figure, retron_source_selected_filename

@app.cell(hide_code=True)
def _(
    dataframe_to_csv_bytes,
    download_safe_stem,
    figure_to_download_bytes,
    mo,
    retron_source_plot_result,
    retron_source_plot_selector,
    retron_source_selected_figure,
    retron_source_selected_source,
    retron_source_supporting_table,
):
    retron_source_export_panel = None
    if (
        retron_source_plot_result is not None
        and retron_source_plot_selector is not None
        and retron_source_selected_source is not None
        and retron_source_supporting_table is not None
    ):
        retron_source_export_stem = download_safe_stem(
            f"{retron_source_selected_source.label}__{retron_source_plot_selector.value}"
        )
        retron_source_export_items = [
            mo.download(
                data=lambda df=retron_source_supporting_table: dataframe_to_csv_bytes(df),
                filename=f"{retron_source_export_stem}__supporting_table.csv",
                mimetype="text/csv",
                label="Table CSV",
            )
        ]
        if retron_source_selected_figure is not None:
            retron_source_export_items = [
                mo.download(
                    data=lambda fig=retron_source_selected_figure: figure_to_download_bytes(
                        fig,
                        fmt="png",
                    ),
                    filename=f"{retron_source_export_stem}.png",
                    mimetype="image/png",
                    label="Figure PNG",
                ),
                mo.download(
                    data=lambda fig=retron_source_selected_figure: figure_to_download_bytes(
                        fig,
                        fmt="svg",
                    ),
                    filename=f"{retron_source_export_stem}.svg",
                    mimetype="image/svg+xml",
                    label="Figure SVG",
                ),
                mo.download(
                    data=lambda fig=retron_source_selected_figure: figure_to_download_bytes(
                        fig,
                        fmt="pdf",
                    ),
                    filename=f"{retron_source_export_stem}.pdf",
                    mimetype="application/pdf",
                    label="Figure PDF",
                ),
                *retron_source_export_items,
            ]
        retron_source_export_panel = mo.hstack(
            retron_source_export_items,
            justify="start",
            wrap=True,
            gap=0.5,
        )
    return (retron_source_export_panel,)


@app.cell(hide_code=True)
def _(
    mo,
    retron_source_contextual_copy,
    retron_source_context_tabs,
    retron_source_control_panel,
    retron_source_export_panel,
    retron_source_plot_error,
    retron_source_plot_result,
    retron_source_selected_figure,
    retron_source_selected_source,
    retron_source_supporting_table_preview,
    retron_source_surface_error,
    retron_source_supporting_table,
    retron_table_kwargs,
):
    retron_source_review_blocks = []
    if retron_source_context_tabs is not None:
        retron_source_review_blocks.append(retron_source_context_tabs)
    retron_source_review_blocks.append(retron_source_control_panel)
    if retron_source_surface_error:
        retron_source_review_blocks.append(
            mo.callout(
                mo.md(
                    "This source experiment cannot be opened as a scoped review surface from the "
                    "master dashboard.\n\n"
                    f"{retron_source_surface_error}"
                ),
                kind="warn",
            )
        )
    elif retron_source_plot_error:
        retron_source_review_blocks.append(
            mo.callout(
                mo.md(retron_source_plot_error),
                kind="warn",
            )
        )
    elif retron_source_plot_result is None:
        retron_source_review_blocks.append(
            mo.callout(
                mo.md("No source-scoped plot is available for the current selection."),
                kind="warn",
            )
        )
    else:
        if retron_source_selected_figure is None:
            retron_source_review_blocks.append(
                mo.callout(
                    mo.md("The selected source plot did not resolve a figure."),
                    kind="warn",
                )
            )
        else:
            retron_source_review_blocks.append(retron_source_selected_figure)
        retron_source_review_blocks.append(
            mo.callout(
                mo.vstack(
                    [
                        mo.md(retron_source_plot_result.title),
                        mo.md(f"Use this view for: {retron_source_contextual_copy['question']}"),
                        mo.md(f"Computed from: `{retron_source_contextual_copy['math']}`"),
                        mo.md(f"Interpretation: {retron_source_contextual_copy['meaning']}"),
                    ],
                    gap=0.08,
                ),
                kind="info",
            )
        )
        if retron_source_export_panel is not None:
            retron_source_review_blocks.append(retron_source_export_panel)
        if retron_source_supporting_table is not None:
            retron_source_review_blocks.append(
                mo.ui.table(
                    retron_source_supporting_table_preview,
                    **retron_table_kwargs(
                        page_size=15,
                        max_height=420,
                    ),
                )
            )
    retron_source_review_panel = mo.vstack(retron_source_review_blocks, gap=0.45)
    return (retron_source_review_panel,)
