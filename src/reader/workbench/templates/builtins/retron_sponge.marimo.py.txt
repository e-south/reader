@app.cell(hide_code=True)
def _(mo, pl, retron_df):
    retron_filter_specs = [
        ("Metric", "metric"),
        ("Sensor", "sensor"),
        ("Sponge", "sponge"),
        ("Stress", "stress_condition"),
    ]
    retron_filter_widgets = {}
    retron_filter_blocks = [mo.md("## Semantic filters")]
    _retron_columns = list(retron_df.columns) if hasattr(retron_df, "columns") else []

    for _retron_label, _retron_column in retron_filter_specs:
        if _retron_column not in _retron_columns:
            continue
        _retron_values = retron_df.select(pl.col(_retron_column).cast(pl.Utf8).drop_nulls().unique()).to_series().to_list()
        _retron_options = ["All"] + sorted(str(value) for value in _retron_values if value is not None)
        retron_filter_widgets[_retron_column] = mo.ui.dropdown(
            options=_retron_options,
            value="All",
            label=_retron_label,
            full_width=True,
        )
        retron_filter_blocks.append(retron_filter_widgets[_retron_column])

    if not retron_filter_widgets:
        retron_filter_blocks.append(mo.md("No semantic filter columns were found in the selected dataset."))

    retron_filter_panel = mo.vstack(retron_filter_blocks)
    return retron_filter_panel, retron_filter_widgets


@app.cell(hide_code=True)
def _(pl, retron_df, retron_filter_widgets):
    retron_filtered_df = retron_df
    for _retron_column, _retron_widget in retron_filter_widgets.items():
        _retron_value = _retron_widget.value
        if _retron_value == "All":
            continue
        retron_filtered_df = retron_filtered_df.filter(pl.col(_retron_column).cast(pl.Utf8) == _retron_value)
    return retron_filtered_df

@app.cell(hide_code=True)
def _(mo, pl, retron_filtered_df, retron_selected_record_id):
    _retron_columns = set(retron_filtered_df.columns) if hasattr(retron_filtered_df, "columns") else set()
    retron_ranking_blocks = [
        mo.md(
            "## O_AUC ranking preview\n"
            "`O_AUC` is the sign-corrected induced-effect AUC. "
            "Positive values mean stronger activity in the expected biological direction after matched-control normalization."
        )
    ]

    if retron_selected_record_id != "semantic_metrics/summary":
        retron_ranking_blocks.append(
            mo.md("Select the semantic summary dataset to preview the sign-corrected ranking surface.")
        )
    elif {"metric", "value", "sensor", "sponge"} <= _retron_columns:
        retron_ranking_df = retron_filtered_df.filter(pl.col("metric") == "O_AUC")
        if "relevant_sensor_pair" in retron_ranking_df.columns:
            retron_ranking_df = retron_ranking_df.filter(pl.col("relevant_sensor_pair").fill_null(False))
        if "sponge" in retron_ranking_df.columns:
            retron_ranking_df = retron_ranking_df.filter(pl.col("sponge") != "tetO")
        retron_keep = [
            retron_column
            for retron_column in ("sensor", "sponge", "stress_condition", "value", "sponge_family_size")
            if retron_column in retron_ranking_df.columns
        ]
        retron_ranking_df = retron_ranking_df.sort("value", descending=True).select(retron_keep).head(12)
        retron_ranking_blocks.append(mo.ui.table(retron_ranking_df, page_size=12))
    else:
        retron_ranking_blocks.append(
            mo.md("The selected dataset does not expose the metric/value columns needed for O_AUC ranking.")
        )

    retron_ranking_panel = mo.vstack(retron_ranking_blocks)
    return retron_ranking_panel
