@app.cell(hide_code=True)
def _(load_cached_parquet_frame, retron_scope_plot_selector, retron_scope_plot_spec_map, retron_scope_record_info):
    retron_scope_selected_spec = (
        retron_scope_plot_spec_map.get(retron_scope_plot_selector.value)
        if retron_scope_plot_selector is not None
        else None
    )
    retron_scope_plot_datasets = {}
    retron_scope_plot_dataset_errors = []
    if retron_scope_selected_spec is not None:
        retron_scope_records_by_id = {
            str(retron_scope_info.get("record_id")): retron_scope_info
            for retron_scope_info in retron_scope_record_info.values()
        }
        for retron_scope_ref in dict(retron_scope_selected_spec.get("reads") or {}).values():
            _retron_scope_record_id = str((retron_scope_ref or {}).get("record", "")).strip()
            if not _retron_scope_record_id or _retron_scope_record_id in retron_scope_plot_datasets:
                continue
            retron_scope_record_meta = retron_scope_records_by_id.get(_retron_scope_record_id)
            if retron_scope_record_meta is None:
                retron_scope_plot_dataset_errors.append(
                    f"Missing dataframe record `{_retron_scope_record_id}` for the selected plot."
                )
                continue
            try:
                retron_scope_plot_datasets[_retron_scope_record_id] = load_cached_parquet_frame(
                    retron_scope_record_meta["path"]
                )
            except Exception as retron_scope_exc:
                retron_scope_plot_dataset_errors.append(
                    f"Failed to load `{_retron_scope_record_id}`: {retron_scope_exc}"
                )
    return retron_scope_plot_dataset_errors, retron_scope_plot_datasets, retron_scope_selected_spec


@app.cell(hide_code=True)
def _(render_retron_experiment_plot, retron_scope_plot_dataset_errors, retron_scope_plot_datasets, retron_scope_selected_spec):
    retron_scope_plot_error = None
    retron_scope_plot_result = None
    if retron_scope_selected_spec is not None and not retron_scope_plot_dataset_errors:
        try:

@app.cell(hide_code=True)
def _(
    mo,
    retron_figure_label,
    retron_scope_contextual_copy,
    retron_scope_figure_selector,
    retron_scope_plot_dataset_errors,
    retron_scope_plot_error,
    retron_scope_plot_labels,
    retron_scope_plot_result,
    retron_scope_plot_selector,
    retron_scope_selected_filename,
    retron_scope_selected_figure,
    retron_scope_supporting_table_preview,
    retron_scope_supporting_table_view,
    retron_scope_export_panel,
    retron_table_kwargs,
):
    retron_scope_review_blocks = [
    ]
    if not retron_scope_plot_labels or retron_scope_plot_selector is None:
        retron_scope_review_blocks.append(mo.md("No plot specs are selected for this experiment."))
    else:
        retron_scope_control_items = [retron_scope_plot_selector]
        if retron_scope_figure_selector is not None:
            retron_scope_control_items.append(retron_scope_figure_selector)
        retron_scope_review_blocks.append(
            mo.hstack(
                retron_scope_control_items,
                gap=0.75,
                align="start",
                justify="start",
                wrap=True,
            )
        )
        if retron_scope_plot_dataset_errors:
            retron_scope_review_blocks.append(
                mo.callout(
                    mo.md("\n".join(f"- {item}" for item in retron_scope_plot_dataset_errors)),
                    kind="warn",
                )
            )
        elif retron_scope_plot_error:
            retron_scope_review_blocks.append(
                mo.callout(
                    mo.md(
                        f"Failed to render the selected plot: `{retron_scope_plot_error}`"
                    ),
                    kind="danger",
                )
            )
        elif retron_scope_plot_result is None:
            retron_scope_review_blocks.append(mo.md("Select a plot view to render."))
        else:
            if retron_scope_figure_selector is not None:
                retron_scope_review_blocks.append(
                    mo.md(f"Figure: {retron_figure_label(retron_scope_selected_filename or '')}")
                )
            if retron_scope_selected_figure is not None:
                retron_scope_review_blocks.append(retron_scope_selected_figure)
            else:
                retron_scope_review_blocks.append(mo.md("This plot spec did not yield a figure."))
            retron_scope_review_blocks.append(
                mo.callout(
                    mo.vstack(
                        [
                            mo.md(f"### {retron_scope_plot_result.title}"),
                            mo.md(f"**Question**\n{retron_scope_contextual_copy['question']}"),
                            mo.md(f"**Formula**\n```text\n{retron_scope_contextual_copy['math']}\n```"),
                            mo.md(f"**Interpretation**\n{retron_scope_contextual_copy['meaning']}"),
                            mo.md(f"**Data**\n`{retron_scope_plot_result.source_record}`"),
                        ],
                        gap=0.08,
                    ),
                    kind="info",
                )
            )
            if retron_scope_export_panel is not None:
                retron_scope_review_blocks.append(retron_scope_export_panel)
            retron_scope_review_blocks.append(
                mo.ui.table(
                    retron_scope_supporting_table_preview,
                    **retron_table_kwargs(
                        page_size=8,
                        max_height=420,
                    ),
                )
            )
    retron_scope_review_panel = mo.vstack(retron_scope_review_blocks, gap=0.45)
    return (retron_scope_review_panel,)
