@app.cell(hide_code=True)
def _(Path, build_notebook_overview, load_notebook_context):
    notebook_context = load_notebook_context(Path(__file__).resolve())
    experiment = notebook_context.experiment
    outputs_dir = notebook_context.outputs_dir
    notebooks_dir = notebook_context.notebooks_dir
    notebook_overview = build_notebook_overview(
        experiment_id=experiment.identity.id,
        experiment_title=experiment.identity.title or "",
        protocol_id=experiment.identity.protocol,
        experiment_root=notebook_context.experiment_root,
        outputs_dir=outputs_dir,
        notebooks_dir=notebooks_dir,
        pipeline_step_ids=notebook_context.pipeline_step_ids,
    )
    return experiment, notebook_overview, outputs_dir, notebooks_dir
